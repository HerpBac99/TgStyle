# FastVLM Server Configuration - Detailed Example
# Подробный пример конфигурации FastVLM сервера со всеми параметрами
# Скопируйте этот файл как .env и настройте под свои нужды

# ==============================================================================
# === ОСНОВНЫЕ НАСТРОЙКИ МОДЕЛИ ===
# ==============================================================================

# Выбор модели: '1.5b', '7b', '7b-int4'
# 1.5b - Быстрая модель (оптимизирована для скорости, меньше памяти)
# 7b - Большая модель (высокое качество, больше памяти, медленнее)
# 7b-int4 - 7B модель с 4-bit квантизацией (баланс качества и скорости)
FASTVLM_MODEL=7b

# ==============================================================================
# === НАСТРОЙКИ СЕРВЕРА ===
# ==============================================================================

# IP адрес сервера (обычно localhost для локальной разработки)
FASTVLM_HOST=127.0.0.1

# Порт сервера (3001 для 1.5B, 3002 для 7B обычно)
FASTVLM_PORT=3001

# ==============================================================================
# === НАСТРОЙКИ МНОГОПОТОЧНОСТИ ===
# ==============================================================================

# Количество рабочих потоков сервера
# Для 1.5B: 8 потоков (быстрее, меньше памяти)
# Для 7B: 4 потока (медленнее, но стабильнее из-за памяти)
FASTVLM_THREADS=8

# Максимальное количество одновременных соединений
# Для 1.5B: 1024 соединений
# Для 7B: 512 соединений (меньше для стабильности)
FASTVLM_CONNECTION_LIMIT=1024

# Таймаут соединений в секундах
# Для 1.5B: 60 секунд
# Для 7B: 120 секунд (больше времени на обработку)
FASTVLM_CONNECTION_TIMEOUT=60

# ==============================================================================
# === НАСТРОЙКИ ГЕНЕРАЦИИ ТЕКСТА (ОЧЕНЬ ВАЖНЫЕ!) ===
# ==============================================================================

# Максимальное количество генерируемых токенов (слов)
# Больше = длиннее ответ, но медленнее и больше памяти
# Для одежды обычно 512-1024 достаточно
MAX_NEW_TOKENS=2048

# ТЕМПЕРАТУРА - главный параметр креативности и точности
# 0.0 = Максимальная точность, повторяющиеся ответы
# 0.1 = Высокая точность, минимальная креативность
# 0.3 = Баланс точности и креативности
# 0.7 = Высокая креативность, менее точные ответы
# 1.0+ = Максимальная креативность, случайные ответы
TEMPERATURE_FASTVLM=0.01

# ДО СЭМПЛИНГ - использовать ли случайный выбор слов
# true = Креативные ответы, разнообразие
# false = Детерминированные ответы, всегда одинаково
DO_SAMPLE=true

# TOP-P (Nucleus Sampling) - доля вероятностей для выбора слов
# 0.1 = Только самые вероятные слова (точнее, но скучнее)
# 0.5 = Средняя креативность
# 0.9 = Высокая креативность, разнообразие
# 0.95 = Максимальное разнообразие
TOP_P=0.95

# TOP-K - количество лучших слов для выбора
# 1 = Только самое вероятное слово (очень точно)
# 50 = Баланс точности и разнообразия
# 100 = Высокое разнообразие
TOP_K=50

# ШТРАФ ЗА ПОВТОРЕНИЯ - предотвращает зацикливание
# 1.0 = Нет штрафа (может повторяться)
# 1.1 = Легкий штраф за повторения
# 1.2 = Средний штраф (рекомендуется)
# 1.5 = Сильный штраф (меньше повторений)
REPETITION_PENALTY=1.2

# КОЛИЧЕСТВО ЛУЧЕЙ (Beam Search) - качество vs скорость/память
# 1 = Быстрее, минимум памяти (~1x для генерации)
# 3 = Баланс качества и скорости (~3x памяти для генерации)
# 5 = Высокое качество, медленнее (~5x памяти для генерации)
# 10 = Максимальное качество, очень медленно (~10x памяти для генерации)
# ПАМЯТЬ: Каждый beam требует отдельный KV-cache, поэтому 3 beams = 3x память на генерацию
NUM_BEAMS=3

# РАННЕЕ ЗАВЕРШЕНИЕ - останавливаться при хорошем ответе
# true = Быстрее, экономит ресурсы
# false = Всегда генерировать максимум токенов
EARLY_STOPPING=true

# ШТРАФ ЗА ДЛИНУ - предпочтение коротких или длинных ответов
# 0.5 = Предпочитает короткие ответы
# 1.0 = Нейтрально (рекомендуется)
# 1.5 = Предпочитает длинные ответы
LENGTH_PENALTY=1.0

# РАЗМЕР N-ГРАММЫ БЕЗ ПОВТОРЕНИЙ - предотвращает повторения фраз
# 2 = Предотвращает повторения 2 слов подряд
# 3 = Предотвращает повторения 3 слов подряд (рекомендуется)
# 4 = Предотвращает повторения 4 слов подряд
NO_REPEAT_NGRAM_SIZE=3

# ==============================================================================
# === НАСТРОЙКИ ПРОИЗВОДИТЕЛЬНОСТИ ===
# ==============================================================================

# Максимальный размер изображения в пикселях (большие стороны)
# 512 = Быстрее, меньше памяти, низкое качество
# 1024 = Баланс качества и скорости
# 2048 = Высокое качество, больше памяти
# 4096 = Максимальное качество, очень много памяти
MAX_IMAGE_SIZE=2048

# Размер батча (количество изображений одновременно)
# 1 = Последовательно (рекомендуется для стабильности)
# 2+ = Параллельно (быстрее, но больше памяти)
BATCH_SIZE=1

# ==============================================================================
# === НАСТРОЙКИ ЛОГИРОВАНИЯ ===
# ==============================================================================

# Уровень логирования
# DEBUG = Все сообщения (для отладки)
# INFO = Основная информация (рекомендуется)
# WARNING = Только предупреждения
# ERROR = Только ошибки
LOG_LEVEL=INFO

# Максимальный размер лог файла в байтах (10MB)
LOG_MAX_BYTES=10485760

# Количество резервных копий лог файлов
LOG_BACKUP_COUNT=5

# ==============================================================================
# === НАСТРОЙКИ GEMINI API (для стилиста) ===
# ==============================================================================

# API ключ Google Gemini (получить на https://makersuite.google.com/app/apikey)
GEMINI_API_KEY=your_gemini_api_key_here

# Модель Gemini для генерации ответов стилиста
# gemini-2.5-flash = Быстрая и умная (рекомендуется)
# gemini-2.5-pro = Более умная, медленнее
# gemini-1.5-flash = Быстрая, менее умная
GEMINI_MODEL=gemini-2.5-flash

# Температура для Gemini (креативность ответов стилиста)
# 0.1 = Консервативные ответы
# 0.7 = Креативные ответы (рекомендуется)
# 1.0 = Очень креативные
GEMINI_TEMPERATURE=0.7

# Максимальное количество токенов для Gemini
GEMINI_MAX_TOKENS=4096

# Бюджет思考 для Gemini (0 = обычный режим)
GEMINI_THINKING_BUDGET=0

# ==============================================================================
# === ДОПОЛНИТЕЛЬНЫЕ ОПТИМИЗАЦИИ ===
# ==============================================================================

# TORCH COMPILE - компиляция модели для ускорения
# true = Быстрее после первого запуска, но дольше загрузка
# false = Стандартный режим (рекомендуется)
TORCH_COMPILE=false

# ==============================================================================
# === НАСТРОЙКИ OLLAMA (альтернатива Gemini) ===
# ==============================================================================

# URL Ollama сервера (если используете локальный Ollama)
OLLAMA_URL=http://127.0.0.1:11434

# Модель Ollama для ответов стилиста
OLLAMA_MODEL=gemma3:4b

# Температура для Ollama стилиста
TEMPERATURE_STYLIST_OLLAMA=0.4

# ==============================================================================
# === РЕКОМЕНДАЦИИ ПО НАСТРОЙКАМ ===
# ==============================================================================

# MAX_NEW_TOKENS=1024
# TOP_P=0.9
# NUM_BEAMS=3

# ДЛЯ ВЫСОКОГО КАЧЕСТВА (7B модель):
# FASTVLM_MODEL=7b
# FASTVLM_THREADS=4
# TEMPERATURE_FASTVLM=0.01
# MAX_NEW_TOKENS=2048
# TOP_P=0.95
# NUM_BEAMS=5
# REPETITION_PENALTY=1.2

# ДЛЯ МИНИМАЛЬНОГО ПОТРЕБЛЕНИЯ ПАМЯТИ:
# MAX_IMAGE_SIZE=1024
# MAX_NEW_TOKENS=512
# NUM_BEAMS=1
# FASTVLM_THREADS=2
