# FastVLM 7B Server Configuration
# Копируйте этот файл в .env и настройте под ваши потребности
# cp env_7b_example.txt .env

# === Настройки сервера ===
FASTVLM7B_HOST=127.0.0.1
FASTVLM7B_PORT=3002

# === Многопоточность (оптимизировано для 7B модели) ===
# Меньше потоков для экономии GPU памяти
FASTVLM7B_THREADS=4
FASTVLM7B_CONNECTION_LIMIT=512
FASTVLM7B_CONNECTION_TIMEOUT=120

# === Настройки генерации FastVLM 7B ===
# Максимальное количество новых токенов (2048 оптимально для 7B)
MAX_NEW_TOKENS_7B=2048

# Температура генерации (0.1 = более точные ответы, 0.7 = более творческие)
TEMPERATURE_7B=0.1

# Использовать sampling (true рекомендуется)
DO_SAMPLE_7B=true

# Top-p sampling (0.8 = сбалансированная точность)
TOP_P_7B=0.8

# Штраф за повторения (1.05 = минимальный штраф)
REPETITION_PENALTY_7B=1.05

# === Настройки производительности ===
# Максимальный размер изображения в пикселях (2048 для 7B модели)
MAX_IMAGE_SIZE_7B=2048

# Размер батча (1 для стабильности 7B модели)
BATCH_SIZE_7B=1

# === Оптимизации PyTorch ===
# Torch compile для ускорения (может потребовать много времени на первый запуск)
TORCH_COMPILE_7B=false

# === Настройки логирования ===
LOG_LEVEL_7B=INFO
LOG_MAX_BYTES_7B=10485760
LOG_BACKUP_COUNT_7B=5

# === Google Gemini API (резервный анализ) ===
# Получите API ключ на https://aistudio.google.com/app/apikey
# GEMINI_API_KEY=your_gemini_api_key_here
# GEMINI_MODEL=gemini-2.5-flash
# GEMINI_TEMPERATURE=0.7
# GEMINI_MAX_TOKENS=4096
# GEMINI_THINKING_BUDGET=0

# === Настройки квантизации ===
# Автоматически включается для GPU < 12GB, но можно принудительно настроить

# Принудительно включить 4-bit квантизацию (экономит ~4GB памяти)
# FORCE_4BIT_QUANTIZATION=true

# Принудительно включить 8-bit квантизацию (экономит ~2GB памяти)  
# FORCE_8BIT_QUANTIZATION=true

# Отключить автоматическую квантизацию (только для мощных GPU)
# DISABLE_AUTO_QUANTIZATION=false

# === Настройки для слабых GPU (6-8GB) ===
# Раскомментируйте если GPU память меньше 8GB

# Уменьшить максимальные токены
# MAX_NEW_TOKENS_7B=1024

# Уменьшить размер изображения
# MAX_IMAGE_SIZE_7B=1024

# Использовать меньше потоков
# FASTVLM7B_THREADS=2

# Уменьшить лимит соединений
# FASTVLM7B_CONNECTION_LIMIT=256

# === Настройки для мощных GPU ===
# Раскомментируйте если GPU память больше 16GB

# Больше токенов для детального анализа
# MAX_NEW_TOKENS_7B=4096

# Больше потоков
# FASTVLM7B_THREADS=8

# Больше соединений
# FASTVLM7B_CONNECTION_LIMIT=1024

# Включить torch compile
# TORCH_COMPILE_7B=true

# === ВАЖНЫЕ ЗАМЕТКИ ===
#
# 1. GPU ПАМЯТЬ:
#    - Минимум 12GB для стабильной работы 7B модели
#    - 16GB+ рекомендуется для комфортной работы
#    - При нехватке памяти уменьшите MAX_NEW_TOKENS_7B и MAX_IMAGE_SIZE_7B
#
# 2. CPU РЕЖИМ:
#    - Если GPU недоступен, модель будет работать на CPU (очень медленно)
#    - Рекомендуется увеличить FASTVLM7B_CONNECTION_TIMEOUT до 300+ секунд
#
# 3. ПРОИЗВОДИТЕЛЬНОСТЬ:
#    - Первый запрос может быть медленным (загрузка модели в память)
#    - Последующие запросы обрабатываются быстрее
#    - Время ответа: 2-10 секунд в зависимости от железа
#
# 4. МОНИТОРИНГ:
#    - Используйте nvidia-smi для мониторинга GPU
#    - Эндпоинт /gpu для получения информации о состоянии
#    - Эндпоинт /load для проверки нагрузки системы
#
# 5. ОПТИМИЗАЦИИ:
#    - Flash Attention ускоряет работу на современных GPU
#    - Torch compile может ускорить генерацию (медленная первая загрузка)
#    - Gradient checkpointing экономит память за счет скорости
